# 3D — Remotion 4 Reference

## Option A: CSS 3D Transforms (Lightweight — Prefer This)

No extra packages. Uses GPU-composited CSS transforms. Sufficient for most motion graphics.

```tsx
import { spring, interpolate, useCurrentFrame, useVideoConfig, AbsoluteFill } from "remotion";

export const CSS3DCard: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();

  const rotateY = interpolate(frame, [0, 120], [0, 360]); // continuous spin
  const rotateX = spring({ frame, fps, config: { damping: 12, stiffness: 60 } });
  const rotateXDeg = interpolate(rotateX, [0, 1], [20, 0]);
  const z = interpolate(rotateX, [0, 1], [-200, 0]);

  return (
    <AbsoluteFill style={{ perspective: "1200px", perspectiveOrigin: "50% 50%" }}>
      <AbsoluteFill style={{ justifyContent: "center", alignItems: "center" }}>
        <div
          style={{
            width: 400,
            height: 240,
            background: "linear-gradient(135deg, #1a1a2e, #16213e)",
            border: "1px solid #f5a623",
            borderRadius: 16,
            boxShadow: "0 40px 80px rgba(0,0,0,0.6)",
            display: "flex",
            justifyContent: "center",
            alignItems: "center",
            fontSize: 48,
            color: "#fff",
            fontWeight: 700,
            transformStyle: "preserve-3d",
            transform: `rotateY(${rotateY}deg) rotateX(${rotateXDeg}deg) translateZ(${z}px)`,
          }}
        >
          INDIA
        </div>
      </AbsoluteFill>
    </AbsoluteFill>
  );
};
```

## CSS 3D Transform Reference

```ts
// Core properties — set on parent for depth context
perspective: "800px"        // lower = more dramatic. 800–1200px is natural
perspectiveOrigin: "50% 50%" // vanishing point
transformStyle: "preserve-3d" // on the element itself for nested 3D

// Transform functions
`rotateX(${deg}deg)`   // tilt forward/back
`rotateY(${deg}deg)`   // spin left/right
`rotateZ(${deg}deg)`   // in-plane rotation (same as rotate())
`translateZ(${px}px)`  // move toward/away from camera
`scale3d(x, y, z)`     // 3D scale
```

## Camera Dolly Move (CSS)

```tsx
const frame = useCurrentFrame();
const { fps } = useVideoConfig();

const dolly = spring({ frame, fps, config: { damping: 20, stiffness: 60 } });
const z = interpolate(dolly, [0, 1], [-300, 0]); // push in from far
const scale = interpolate(dolly, [0, 1], [0.7, 1]);

<AbsoluteFill style={{ perspective: "1000px" }}>
  <div style={{ transform: `translateZ(${z}px) scale(${scale})` }}>
    {/* scene content */}
  </div>
</AbsoluteFill>
```

## Option B: `@remotion/three` (Three.js — Full 3D)

Use when you need actual 3D geometry, lighting, or shadows.

```tsx
import { ThreeCanvas } from "@remotion/three";
import { useCurrentFrame } from "remotion";
import { useRef } from "react";
import * as THREE from "three";

const RotatingBox: React.FC = () => {
  const frame = useCurrentFrame();
  // frame is available here — no hooks needed, just use the prop
  const rotY = (frame / 30) * Math.PI * 2; // full rotation every 30 frames

  return (
    <mesh rotation={[0.3, rotY, 0]}>
      <boxGeometry args={[2, 2, 2]} />
      <meshStandardMaterial color="#f5a623" metalness={0.3} roughness={0.4} />
    </mesh>
  );
};

export const ThreeScene: React.FC = () => {
  const frame = useCurrentFrame();
  return (
    <ThreeCanvas width={1920} height={1080}>
      {/* Lighting */}
      <ambientLight intensity={0.4} />
      <directionalLight position={[5, 5, 5]} intensity={1.2} />

      {/* Camera is set by ThreeCanvas — override with PerspectiveCamera if needed */}

      {/* Geometry */}
      <RotatingBox />
    </ThreeCanvas>
  );
};
```

## Animating Three.js with `useCurrentFrame()`

```tsx
// Inside a @react-three/fiber component, access frame via prop (not hook)
// Pass frame down from the Remotion component

const AnimatedSphere: React.FC<{ frame: number }> = ({ frame }) => {
  const y = Math.sin(frame * 0.1) * 1.5; // bounce on sine wave
  return (
    <mesh position={[0, y, 0]}>
      <sphereGeometry args={[1, 32, 32]} />
      <meshStandardMaterial color="#e94560" />
    </mesh>
  );
};
```

## Camera Dolly in Three.js

```tsx
import { PerspectiveCamera } from "@react-three/drei";

const DollyCamera: React.FC<{ frame: number }> = ({ frame }) => {
  const z = interpolate(frame, [0, 60], [10, 5], { extrapolateRight: "clamp" });
  return <PerspectiveCamera makeDefault position={[0, 0, z]} fov={60} />;
};
```

## Performance Tips

- Keep polygon count under 50k total for smooth 30fps render in Remotion
- Avoid `MeshPhysicalMaterial` — `MeshStandardMaterial` is 90% as good, much faster
- Prefer CSS 3D for flat-plane animations (cards, text, panels) — no THREE overhead
- Avoid post-processing (`@react-three/postprocessing`) — costly in Remotion's frame-by-frame render
- For text in Three.js, use `@react-three/drei Text` — simpler than TextGeometry

## Gotchas

- `ThreeCanvas` takes `width`/`height` in pixels, NOT CSS — always pass exact composition dimensions.
- `useCurrentFrame()` cannot be called inside Three.js components (they're not React components in the Remotion sense). Pass `frame` as a prop.
- CSS `perspective` is applied to the parent. The 3D-transformed element needs `transformStyle: "preserve-3d"` only if it has 3D children.
- `rotateY` in CSS is degrees; in Three.js it is radians.
